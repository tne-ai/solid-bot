Quickstart

  

    
1. Discover — fetch agent card

    
curl https://solid-bot.dev.tne.ai/.well-known/agent-card.json

  

  

    
2. Call a free skill — A2A JSON-RPC format

    
const res = await fetch('https://solid-bot.dev.tne.ai/a2a/jsonrpc', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    jsonrpc: '2.0', method: 'message/send', id: 1,
    params: { message: { kind: 'message', parts: [
      { kind: 'text', text: 'Is Solid a good fit for a health app where patients own their records?' }
    ]}}
  })
});
const { result } = await res.json();
console.log(result.parts[0].text);

  

  

    
3. Paid skill — @x402/fetch handles 402 automatically

    
// npm install @x402/fetch @x402/evm viem
import fs from 'fs/promises';
import { wrapFetchWithPayment, x402Client } from '@x402/fetch';
import { ExactEvmScheme } from '@x402/evm';
import { privateKeyToAccount } from 'viem/accounts';

// Agent's funded wallet (needs USDC on Base Sepolia)
const account = privateKeyToAccount(process.env.EVM_PRIVATE_KEY);
const client = new x402Client();
client.register('eip155:*', new ExactEvmScheme(account));

// Wrap fetch — 402 → sign → retry happens automatically
const fetch402 = wrapFetchWithPayment(globalThis.fetch, client);

const res = await fetch402('https://solid-bot.dev.tne.ai/api/scaffold', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    topic: 'A health app where patients own their records',
    components: ['auth', 'pod-reader', 'pod-writer']  // $0.30 USDC
  })
});

const { files } = await res.json();
// files[] — write directly to disk
for (const { filename, content } of files) {
  await fs.writeFile(filename, content);
}
